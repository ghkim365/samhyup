2. 구글 스프레드시트 mail_log 연동 및 수식

  • 메인 company 시트의 M, N, O, P열에 아래 수식을 각각 적용하여 최종 발송일과 발송 경과를 완전 자동 추적하도록 구성합니다:
  • M열 (발송대기): 직접 발송 선택 (체크박스/대기)
    N열 (발송상태): 발송 시 GAS에 의해 성공/실패 기록
    최종 발송일 (O2열): =IFERROR(MAXIFS(mail_log!C:C, mail_log!D:D, C2), "")
    발송 경과일 (P2열): =IF(ISBLANK(O2), "", TODAY() - O2)
  • 조건부 서식 지정: P열을 전체 드래그하여 조건부 서식 -> 셀 강조 규칙 -> 보다 큼 -> 9일 경우 주황색 또는 빨간색 채우기를 활성화합니다. (10일 단위로 후속 메일을 발송할 대상을 자동으로 모니터링하기 위함입니다)

GAS 수정 코드 — doGet() 함수 (자동 연동용)

function doGet(e) {
  var site = e.parameter.site;
  var cb = e.parameter.callback;
  var data = {};

  if (site === 'mailling') {
    var ss = SpreadsheetApp.openById(SS_ID);
    var logSheet = ss.getSheetByName("mail_log");
    var logs = [];
    var visits = { success: 0, fail: 0 };

    if (logSheet) {
      var rows = logSheet.getDataRange().getValues();
      for (var i = rows.length - 1; i >= 1; i--) {
        var s = rows[i][6]; // G열: 발송상태
        if (s === '성공') visits.success++;
        else if (s === '실패') visits.fail++;
        logs.push({
          id: rows[i][1], date: rows[i][2], company: rows[i][3],
          email: rows[i][4], round: rows[i][5], status: s,
          detail: rows[i][7], note: rows[i][8] || ''
        });
      }
    }
    data = { visits: visits, logs: logs };
  }

  return ContentService
    .createTextOutput(cb + "(" + JSON.stringify(data) + ")")
    .setMimeType(ContentService.MimeType.JAVASCRIPT);
}
